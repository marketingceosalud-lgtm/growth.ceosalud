function doLogin() { var email = (document.getElementById('login-email') || {value:''}).value.trim(); var pass = (document.getElementById('login-pass') || {value:''}).value; var btn = document.getElementById('login-btn'); var errEl = document.getElementById('login-error'); if (!email || !pass) { showLoginError('Por favor ingresa tu correo y contraseña.'); return; } if (btn) { btn.textContent = 'Verificando...'; btn.disabled = true; } if (errEl) errEl.style.display = 'none'; // Intento de autenticación directa con Firebase auth.signInWithEmailAndPassword(email, pass) .then(function(cred) { console.log("Acceso exitoso:", cred.user.email); // Ocultar pantalla de login y mostrar ERP de inmediato document.getElementById('login-screen').style.display = 'none'; document.getElementById('erp-app').style.display = 'block'; var perfil = ROLES[cred.user.email] || { rol: 'admin', nombre: cred.user.email }; actualizarSessionUI(perfil); filtrarSidebarPorRol(perfil.rol); go('dash'); showToast('Bienvenido, ' + perfil.nombre); }) .catch(function(err) { console.error("Error Firebase Auth:", err.code, err.message); if (btn) { btn.textContent = 'Ingresar al sistema'; btn.disabled = false; } var msg = 'Error de acceso: ' + err.message; if (err.code === 'auth/operation-not-allowed') { msg = 'Atención: Debes habilitar "Correo electrónico/contraseña" en Firebase Console > Authentication > Método de acceso.'; } else if (err.code === 'auth/invalid-credential' || err.code === 'auth/wrong-password' || err.code === 'auth/user-not-found') { msg = 'Usuario o contraseña incorrectos. Verifica que el usuario exista en Firebase Authentication.'; } else if (err.code === 'auth/invalid-email') { msg = 'El formato del correo es inválido (asegúrate de que termine en @ceosalud.pe).'; } showLoginError(msg); }); }
